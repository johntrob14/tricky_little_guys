def second_largest(values: Iterable[T]) -> Optional[T]:
    """
    Return the second largest *distinct* element from *values*.

    Parameters
    ----------
    values : iterable of comparable items
        The source of numbers (or any objects that support the ``>`` operator).

    Returns
    -------
    Optional[T]
        The second largest distinct element, or ``None`` if there are fewer
        than two distinct values.

    Examples
    --------
    >>> second_largest([1, 2, 3, 4])
    3
    >>> second_largest([5, 5, 5, 5])
    >>> second_largest([7, 2, 9, 9, 4])
    7
    >>> second_largest([])
    >>> second_largest([42])
    """

    # Convert to a set to discard duplicate entries.
    # Using a set also guarantees O(1) look‑ups for the “already seen” check.
    distinct = set(values)

    # If we have fewer than two different numbers, we cannot determine a
    # second largest value.
    if len(distinct) < 2:
        return None

    # The most straightforward way is to pop the maximum, then the new
    # maximum is the second largest.
    # This avoids a full sort (which would be O(n log n)) and runs in O(n)
    # average time, O(n) worst‑case, with O(1) extra space.
    # (The built‑in ``max`` scans the container once.)
    largest = max(distinct)
    distinct.remove(largest)          # discard the top element
    second_largest = max(distinct)    # now the biggest remaining element

    return second_largest
